   640→  async function handleLogin(ev: FormEvent<HTMLFormElement>) {
   641→    ev.preventDefault()
   642→    setLoginError(null)
   643→    if (!MOBILE_RE.test(loginValues.mobile.trim())) {

   645→      return
   646→    }
   647→    if (!loginValues.password) {
   648→      setLoginError('Password is required')
   649→      return
   650→    }
   651→    setLoggingIn(true)
   652→    try {
   653→      const res = await fetch('/api/auth/login', {
   654→        method: 'POST',
   655→        headers: { 'Content-Type': 'application/json' },
   656→        body: JSON.stringify({
   657→          mobile: loginValues.mobile.trim(),
   658→          password: loginValues.password,
   659→        }),
   660→      })
   661→      const data = await res.json()
   662→      if (!res.ok) {
   663→        setLoginError(data.error ?? 'Login failed. Please try again.')
   664→        return
   665→      }
   666→      setLoggedInUser(data.user?.username ?? '')
   667→      toast({
   668→        title: 'Login successful',
   669→        description: `Welcome back, ${data.user?.username ?? ''}!`,
   670→      })
   671→    } catch {
   672→      setLoginError('Network error. Please try again.')
   673→    } finally {
   674→      setLoggingIn(false)
   675→    }
   676→  }
